import tkinter as tr
from tkinter import messagebox
from tkinter import ttk 

k_dosya = "kullanicilar.txt"
y_dosya = "yorumlar.txt"

bg_color = "#f0f2f5"  
primary_color = "#4a90e2" 
text_color = "#333333"

def sorgula():
    global kullanici
    admin_input = text.get()
    sifre_input = text1.get()
    giris_basarili = False

    try:
        with open(k_dosya, "r", encoding="utf-8") as dosya:
            for satir in dosya:
                satir = satir.strip()
                if "," in satir:
                    kadi, ksifre = satir.split(",")
                    if admin_input == kadi and sifre_input == ksifre:
                        giris_basarili = True
                        break
    except FileNotFoundError:
        messagebox.showerror("Hata", "Kullanıcı veri tabanı bulunamadı!")
        return

    if giris_basarili:
        kullanici = admin_input
        form.destroy()  
        yeni_pencere()  
    else:
        messagebox.showerror("Hata", "Kullanıcı adı veya şifre yanlış!")

def kayit():
    global text4, text5, form3
    form3 = tr.Toplevel()
    form3.title("Yeni Kayıt")
    form3.geometry("350x250")
    form3.configure(bg=bg_color)
    
    tr.Label(form3, text="Yeni Hesap Oluştur", font=("Arial", 12, "bold"), bg=bg_color, fg=primary_color).pack(pady=10)
    
    tr.Label(form3, text="Kullanıcı Adı:", bg=bg_color).pack()
    text4 = tr.Entry(form3, font=("Arial", 10))
    text4.pack(pady=5)
    
    tr.Label(form3, text="Şifre:", bg=bg_color).pack()
    text5 = tr.Entry(form3, show="*", font=("Arial", 10))
    text5.pack(pady=5)
    
    tr.Button(form3, text="Kaydı Tamamla", command=kaydi_tamamla, bg="#2ecc71", fg="white", font=("Arial", 10, "bold"), width=15).pack(pady=15)

def kaydi_tamamla():
    k_adi = text4.get()
    k_sifre = text5.get()
    if k_adi and k_sifre:
        with open(k_dosya, "a", encoding="utf-8") as dosya:
            dosya.write(f"{k_adi},{k_sifre}\n")
        messagebox.showinfo("Başarılı", "Kaydınız başarıyla oluşturuldu!")
        form3.destroy()
    else:
        messagebox.showwarning("Uyarı", "Lütfen tüm alanları doldurun!")

def yeni_pencere():
    form2 = tr.Tk()
    form2.title("Yorum Sayfası")
    form2.geometry("400x300")
    form2.configure(bg=bg_color)
    
    tr.Label(form2, text=f"Hoş geldin, {kullanici}", font=("Arial", 14, "bold"), bg=bg_color, fg=text_color).pack(pady=20)
    
    tr.Label(form2, text="Yorumunuzu Yazın:", bg=bg_color).pack()
    text2 = tr.Entry(form2, font=("Arial", 11), width=30)
    text2.pack(pady=10)
    
    def yorum_kaydet():
        icerik = text2.get()
        if icerik:
            with open(y_dosya, "a", encoding="utf-8") as dosya:
                dosya.write(f"{kullanici}: {icerik}\n")
            messagebox.showinfo("Başarılı", "Yorumunuz paylaşıldı!")
            text2.delete(0, tr.END)
        else:
            messagebox.showwarning("Uyarı", "Boş yorum yapamazsınız!")

    tr.Button(form2, text="Gönder", command=yorum_kaydet, bg=primary_color, fg="white", font=("Arial", 10, "bold"), width=20).pack(pady=15)
    form2.mainloop()

def admin_girisi():
    global text6, text7
    form4 = tr.Toplevel()
    form4.title("Yönetici Paneli")
    form4.geometry("300x200")
    form4.configure(bg="#34495e")
    
    tr.Label(form4, text="Admin Girişi", font=("Arial", 11, "bold"), bg="#34495e", fg="white").pack(pady=10)
    
    tr.Label(form4, text="Admin Kullanıcı Adı:", bg="#34495e", fg="white").pack()
    text6 = tr.Entry(form4)
    text6.pack(pady=2)
    
    tr.Label(form4, text="Şifre:", bg="#34495e", fg="white").pack()
    text7 = tr.Entry(form4, show="*")
    text7.pack(pady=2)
    
    tr.Button(form4, text="Giriş Yap", command=admin_kontrol, bg="#e74c3c", fg="white", font=("Arial", 9, "bold")).pack(pady=10)

def admin_kontrol():
    if text6.get() == "admin" and text7.get() == "1234":
        try:
            with open(y_dosya, "r", encoding="utf-8") as f:
                icerik = f.read()
                if not icerik: icerik = "Henüz yorum yapılmamış."
                messagebox.showinfo("Tüm Yorumlar", icerik)
        except FileNotFoundError:
            messagebox.showinfo("Bilgi", "Henüz yorum dosyası oluşturulmamış.")
    else:
        messagebox.showerror("Hata", "Admin yetkisi yok!")

form = tr.Tk()
form.title("Giriş Paneli")
form.geometry("350x400")
form.configure(bg=bg_color)

tr.Label(form, text="HOŞ GELDİNİZ", font=("Arial", 16, "bold"), bg=bg_color, fg=primary_color).pack(pady=30)

tr.Label(form, text="Kullanıcı Adı", font=("Arial", 10), bg=bg_color, fg=text_color).pack()
text = tr.Entry(form, font=("Arial", 11), bd=2, relief="flat")
text.pack(pady=5, ipady=3)

tr.Label(form, text="Şifre", font=("Arial", 10), bg=bg_color, fg=text_color).pack()
text1 = tr.Entry(form, show="*", font=("Arial", 11), bd=2, relief="flat")
text1.pack(pady=5, ipady=3)

style = ttk.Style()
tr.Button(form, text="Giriş Yap", command=sorgula, bg=primary_color, fg="white", font=("Arial", 10, "bold"), width=20, cursor="hand2").pack(pady=15)
tr.Button(form, text="Kayıt Ol", command=kayit, bg="#95a5a6", fg="white", font=("Arial", 10, "bold"), width=20, cursor="hand2").pack(pady=5)
tr.Button(form, text="Admin Paneli", command=admin_girisi, bg="#34495e", fg="white", font=("Arial", 9), width=15, cursor="hand2").pack(side="bottom", pady=20)

form.mainloop()
